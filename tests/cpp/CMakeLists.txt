cmake_minimum_required(VERSION 3.28)
project(ASKSDKTests LANGUAGES C CXX)

# Exercise an installed SDK independently from ASK's own build tree.
set(ASK_TEST_COMPONENT "Compiler" CACHE STRING "SDK component to test")
set_property(CACHE ASK_TEST_COMPONENT PROPERTY STRINGS Compiler Core Runtime)
if(NOT ASK_TEST_COMPONENT MATCHES "^(Compiler|Core|Runtime)$")
  message(FATAL_ERROR "ASK_TEST_COMPONENT must be Compiler, Core or Runtime")
endif()
find_package(ASK CONFIG REQUIRED COMPONENTS ${ASK_TEST_COMPONENT})
enable_testing()

if(ASK_TEST_COMPONENT STREQUAL "Compiler")
  add_executable(ask-compiler-test compiler.cpp)
  target_link_libraries(ask-compiler-test PRIVATE ASK::Compiler ${CMAKE_DL_LIBS})
  add_test(NAME ask-compiler COMMAND ask-compiler-test "${CMAKE_CURRENT_BINARY_DIR}/compiler-test.o")
elseif(ASK_TEST_COMPONENT STREQUAL "Core")
  add_executable(ask-core-test frontend.cpp)
  target_link_libraries(ask-core-test PRIVATE ASK::Core)
  add_test(NAME ask-core COMMAND ask-core-test --optimize)
else()
  add_executable(ask-runtime-test runtime.cpp)
  target_link_libraries(ask-runtime-test PRIVATE ASK::Runtime)
  add_test(NAME ask-runtime COMMAND ask-runtime-test)
endif()
